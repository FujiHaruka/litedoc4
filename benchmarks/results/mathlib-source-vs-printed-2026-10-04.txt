Method
------
- Class "no own source": the source text at the declaration's selection range is not its own
  name preceded by a declaration keyword (def/theorem/lemma/abbrev/instance/structure/class/
  inductive/opaque/axiom/irreducible_def) and not inside an attribute (`@[…]`, `attribute […]`).
  Bucketed by what that text is.
- Header: from the end of the name to the first top-level `:=`, `where`, `deriving`, or a `|`
  starting a line, capped at the declaration's end; comments and strings blanked first.
- A printed binder is "from the header" when one of its names is declared in a header binder
  group or bound by `∀`/`Π` in the written type; an anonymous instance binder when its
  whitespace-normalised text appears in the header (fallback: its identifier set, 7,582 binders).
  Otherwise "not in the header".

Error bounds
------------
- 60 declarations read by hand across the classes: no misclassification seen.
- Counting no `∀`-bound names as header names: 201,737 → 202,315 (+578).
- No identifier-set fallback for instance binders: 201,737 → 202,312 (+575).
- A printed group mixing header and `variable` names counts as header: class 2 is undercounted by
  an unmeasured amount. 727 hygienic `✝` binders count as header.
- Against .ilean as an independent list: 737 own-source declarations missing from it (0.3%);
  1,589 class-1 declarations other than instances present in it (alias 377, field lines 109,
  ≈ 550 notation/syntax/macro).
- Shown "with the same binders" still differs in small ways (`Type*` vs `Type u_4`, `Perm` vs
  `Equiv.Perm`, a missing `↑`); not counted.

import json, glob, os, re, sys

IR = '/private/tmp/lean-doc-relay/wmathlib/out/ir/modules/'
HERE = os.path.dirname(os.path.abspath(__file__))
SRC = os.path.join(HERE, 'src')

OPEN = {'(': ')', '[': ']', '{': '}', '⦃': '⦄', '⟨': '⟩', '⟦': '⟧'}
CLOSE = set(OPEN.values())
BINDER_OPEN = set('([{⦃')
DECL_KW = {'def', 'theorem', 'lemma', 'abbrev', 'instance', 'structure', 'class', 'inductive',
           'opaque', 'axiom', 'example', 'irreducible_def', 'alias', 'coinductive'}
TOK = re.compile(r"«[^»]*»|[\w'!?]+(?:\.(?:«[^»]*»|[\w'!?]+))*|:=|::|\S")


def blank(text):
    out = list(text)
    i, n = 0, len(text)
    while i < n:
        c = text[i]
        if c == '-' and text.startswith('--', i):
            j = text.find('\n', i)
            j = n if j < 0 else j
            for k in range(i, j):
                out[k] = ' '
            i = j
        elif c == '/' and text.startswith('/-', i):
            depth, j = 1, i + 2
            while j < n and depth:
                if text.startswith('/-', j):
                    depth += 1; j += 2
                elif text.startswith('-/', j):
                    depth -= 1; j += 2
                else:
                    j += 1
            for k in range(i, j):
                if out[k] != '\n':
                    out[k] = ' '
            i = j
        elif c == '"':
            j = i + 1
            while j < n and text[j] != '"':
                j += 2 if text[j] == '\\' else 1
            for k in range(i + 1, min(j, n)):
                if out[k] != '\n':
                    out[k] = ' '
            i = j + 1
        else:
            i += 1
    return ''.join(out)


class Module:
    def __init__(self, mod):
        raw = open(os.path.join(SRC, mod.replace('.', '/') + '.lean'), encoding='utf-8').read()
        self.raw = raw
        self.text = blank(raw)
        self.starts = [0]
        for i, c in enumerate(raw):
            if c == '\n':
                self.starts.append(i + 1)

    def off(self, line, col):
        return self.starts[line - 1] + col


def attr_context(text, start, sel):
    depth = 0
    i = sel - 1
    lo = max(0, sel - 3000)
    while i >= lo:
        c = text[i]
        if c in CLOSE:
            depth += 1
        elif c in OPEN:
            if depth == 0:
                if c == '[' and (text[i - 1:i] == '@' or text[max(0, i - 10):i].rstrip().endswith('attribute')):
                    seg = text[i + 1:sel]
                    parts, d, cur = [], 0, ''
                    for ch in seg:
                        if ch in OPEN: d += 1
                        elif ch in CLOSE: d -= 1
                        if ch == ',' and d == 0:
                            parts.append(cur); cur = ''
                        else:
                            cur += ch
                    m = re.match(r"[\w!?]+", (cur + text[sel:sel + 60]).strip())
                    return m.group(0) if m else '?'
            else:
                depth -= 1
        i -= 1
    return None


def prev_kw(text, pos):
    i = pos - 1
    while i >= 0 and text[i].isspace():
        i -= 1
    if i >= 0 and text[i] == ')':
        d = 0
        while i >= 0:
            if text[i] == ')': d += 1
            elif text[i] == '(':
                d -= 1
                if d == 0: break
            i -= 1
        return prev_word(text, i)
    return prev_word(text, pos)


def prev_word(text, pos):
    i = pos - 1
    while i >= 0 and text[i].isspace():
        i -= 1
    j = i
    while j >= 0 and (text[j].isalnum() or text[j] in "_'!?."):
        j -= 1
    return text[j + 1:i + 1]


def header_span(mod, hs, end):
    """From hs, scan to first depth-0 ':=', line-leading '|', or ' where'/'deriving' token, bounded by end."""
    t = mod.text
    depth = 0
    i = hs
    while i < end:
        c = t[i]
        if c in OPEN:
            depth += 1
        elif c in CLOSE:
            depth = max(0, depth - 1)
        elif depth == 0:
            if c == ':' and t.startswith(':=', i):
                return i
            if c == '|':
                ls = t.rfind('\n', 0, i) + 1
                if t[ls:i].strip() == '':
                    return i
            if c in 'wd' and (i == 0 or not (t[i - 1].isalnum() or t[i - 1] in "_'.")):
                for kw in ('where', 'deriving'):
                    if t.startswith(kw, i):
                        k = i + len(kw)
                        if k >= len(t) or not (t[k].isalnum() or t[k] in "_'!?"):
                            return i
        i += 1
    return end


def is_ident(tok):
    return tok[0] == '«' or tok[0].isalnum() or tok[0] in "_'"


def last_comp(tok):
    return tok.replace('«', '').replace('»', '').split('.')[-1]


def first_comp(tok):
    if tok.startswith('«'):
        return tok.split('»')[0] + '»'
    return tok.split('.')[0]


def analyse_header(h):
    """Return (typed, names bound in binding positions, list of [..] group texts, list of [..] group ident-sets)."""
    toks = TOK.findall(h)
    names = set()
    inst_groups = []
    inst_sets = []
    typed = False
    depth = 0
    i = 0
    n = len(toks)
    group = None  # (opener, tokens)
    while i < n:
        tk = toks[i]
        if depth == 0 and tk == ':':
            typed = True
            i += 1
            break
        if tk in OPEN:
            if depth == 0:
                group = (tk, [])
            elif group is not None:
                group[1].append(tk)
            depth += 1
        elif tk in CLOSE:
            depth = max(0, depth - 1)
            if depth == 0 and group is not None:
                take_group(group, names, inst_groups, inst_sets)
                group = None
            elif group is not None:
                group[1].append(tk)
        elif group is not None:
            group[1].append(tk)
        i += 1
    if typed:
        rest = toks[i:]
        for j, tk in enumerate(rest):
            if tk in ('∀', 'Π') and not os.environ.get('NOFORALL'):
                forall_names(rest, j + 1, names, inst_groups, inst_sets)
    return typed, names, inst_groups, inst_sets


def take_group(group, names, inst_groups, inst_sets):
    opener, toks = group
    if opener not in BINDER_OPEN:
        return
    d = 0
    colon = None
    for k, tk in enumerate(toks):
        if tk in OPEN:
            d += 1
        elif tk in CLOSE:
            d -= 1
        elif d == 0 and tk == ':':
            colon = k
            break
    if opener == '[':
        inst_groups.append(norm('[' + ' '.join(toks) + ']'))
        ids = [t for t in toks if is_ident(t)]
        if ids:
            inst_sets.append((last_comp(ids[0]), frozenset(p for t in ids for p in t.replace('«', '').replace('»', '').split('.'))))
        if colon is None:
            return
    lead = toks[:colon] if colon is not None else toks
    for tk in lead:
        if is_ident(tk):
            names.add(tk)
        else:
            break


def forall_names(toks, j, names, inst_groups, inst_sets):
    depth = 0
    pos_ok = True
    group = None
    while j < len(toks):
        tk = toks[j]
        if depth == 0 and tk == ',':
            return
        if tk in OPEN:
            if depth == 0:
                group = (tk, [])
            elif group is not None:
                group[1].append(tk)
            depth += 1
        elif tk in CLOSE:
            depth -= 1
            if depth < 0:
                return
            if depth == 0 and group is not None:
                take_group(group, names, inst_groups, inst_sets)
                group = None
            elif group is not None:
                group[1].append(tk)
        elif group is not None:
            group[1].append(tk)
        elif depth == 0:
            if pos_ok and is_ident(tk):
                names.add(tk)
            else:
                pos_ok = False
        j += 1


def norm(s):
    s = re.sub(r'\s+', ' ', s)
    s = re.sub(r'\s*([\[\]()])\s*', r'\1', s)
    return s.strip()


BTOK = re.compile(r"[^\s(){}\[\]⦃⦄:]+")


def binder_names(b):
    """Names of a printed binder, or None for an anonymous instance binder."""
    inner = b[1:-1] if b[:1] in '({[⦃' else b
    if b.startswith('{{'):
        inner = b[2:-2]
    if ' : ' not in inner:
        if b.startswith('['):
            return None
        return inner.split()
    if b.startswith('['):
        # printed named instance binder "[inst : C]"
        lead = inner.split(' : ', 1)[0]
        return lead.split()
    return inner.split(' : ', 1)[0].split()


def classify_binders(binders, h_names, inst_groups, inst_sets):
    res = []  # per binder: 'header' | 'notin' | 'hyg'
    fallback = 0
    hn = set(h_names) | {first_comp(x) for x in h_names}
    for b in binders:
        ns = binder_names(b)
        if ns is None:
            nb = norm(b)
            if nb in inst_groups:
                res.append('header')
            else:
                ids = [t for t in TOK.findall(b[1:-1]) if is_ident(t)]
                if ids and not os.environ.get('NOFALLBACK') and any(hd == last_comp(ids[0]) and {last_comp(t) for t in ids} <= comps for hd, comps in inst_sets):
                    res.append('header'); fallback += 1
                else:
                    res.append('notin')
            continue
        if all('✝' in x for x in ns):
            res.append('hyg'); continue
        if any(x in hn or x.strip('«»') in hn for x in ns):
            res.append('header')
        else:
            res.append('notin')
    return res, fallback


def bucket_for(mod, x, sel_text, start, sel):
    t = mod.text
    a = attr_context(t, start, sel)
    if a is not None:
        return 'attr:' + a
    pw = prev_word(t, sel)
    first = t[sel:sel + len(sel_text)].strip().split('\n')[0].strip()
    if first.startswith('instance'):
        return 'instance (auto-named)'
    if pw == 'deriving' or re.search(r'\bderiving\b[^\n]*$', t[t.rfind('\n', 0, sel) + 1:sel]):
        return 'deriving'
    if x['kind'] == 'constructor':
        return 'constructor (auto mk)'
    ls = t.rfind('\n', 0, sel) + 1
    line = re.sub(r'^\s*(@\[[^\]]*\]\s*)?((private|protected|local|scoped|noncomputable)\s+)*', '', t[ls:sel + len(sel_text)])
    w = re.match(r"[\w!?']+", line)
    return 'command: ' + (w.group(0) if w else line[:20])


def tsv(fields):
    return '\t'.join(' '.join(str(f).split()) if f else '' for f in fields)


def strip_priv(n):
    return re.sub(r'^_private\.(?:[^.]+\.)*?0\.', '', n)


def run(files, out_tsv, out_spot_src):
    fo = open(out_tsv, 'w')
    fo.write('module\tname\tkind\tcls\tbucket\tkw\tnbind\tnnot\tnhyg\tnfallback\ttyped\tnotin\n')
    fs = open(out_spot_src, 'w')
    for f in files:
        d = json.load(open(f))
        mod = Module(d['module'])
        containers = []
        for y in d['declarations']:
            if y['kind'] in ('structure', 'class', 'inductive') and y.get('selectionRange') and y.get('line') is not None:
                ysr = y['selectionRange']
                containers.append((y['name'], mod.off(ysr[2], ysr[3]), mod.off(y['endLine'], y['endCol']), y['kind']))
        for x in d['declarations']:
            sr = x.get('selectionRange')
            row = [d['module'], x['name'], x['kind']]
            if not sr or x.get('line') is None:
                fo.write(tsv(row + ['norange', '-', '-', str(len(x['binders'])), '', '', '', '', '']) + '\n')
                continue
            start = mod.off(x['line'], x['col'])
            end = mod.off(x['endLine'], x['endCol'])
            s0 = mod.off(sr[0], sr[1]); s1 = mod.off(sr[2], sr[3])
            sel_text = mod.raw[s0:s1]
            last = x['name'].split('.')[-1]
            st = sel_text.replace('«', '').replace('»', '')
            name_ok = st.split('.')[-1] == last.replace('«', '').replace('»', '')
            in_attr = attr_context(mod.text, start, s0)
            kw = prev_kw(mod.text, s0)
            container = next((c for c in containers if c[1] <= s0 < c[2] and strip_priv(x['name']).startswith(strip_priv(c[0]) + '.')), None)
            auto_inst = sel_text.lstrip().startswith('instance') and x['kind'] == 'instance' and not name_ok
            line_pre = mod.text[mod.text.rfind('\n', 0, s0) + 1:s0]
            hs = None; cls = 'gen'
            if in_attr is not None:
                bucket = 'attr:' + in_attr
            elif auto_inst:
                bucket = 'instance (auto-named)'
                hs = s0 + sel_text.index('instance') + len('instance'); kw = 'instance'
            elif kw == 'alias' or re.search(r'\balias\s*⟨', line_pre):
                bucket = 'alias (signature is the target\'s)'
            elif name_ok and kw in DECL_KW:
                cls = 'own'; bucket = 'own'; hs = s1
            elif container is not None:
                if x['kind'] == 'constructor':
                    bucket = 'constructor (named, inside the %s)' % container[3]
                elif kw == 'extends' or re.search(r'\bextends\b', mod.text[container[1]:s0]) and not re.search(r'\bwhere\b', mod.text[container[1]:s0]):
                    bucket = 'parent projection (extends clause)'
                else:
                    bucket = 'field projection (field line inside the %s)' % container[3]
            else:
                bucket = bucket_for(mod, x, sel_text, start, s0)
            if hs is None:
                fo.write(tsv(row + [cls, bucket, kw, str(len(x['binders'])), '', '', '', '', '']) + '\n')
                continue
            he = header_span(mod, hs, max(end, hs))
            h = mod.text[hs:he]
            typed, names, ig, iset = analyse_header(h)
            res, fb = classify_binders(x['binders'], names, ig, iset)
            notin = [b for b, r in zip(x['binders'], res) if r == 'notin']
            fo.write(tsv(row + [cls, bucket, kw, str(len(res)), str(len(notin)), str(res.count('hyg')),
                                      str(fb), 'T' if typed else 'F', ' '.join(' '.join(b.split()) for b in notin)]) + '\n')
            fs.write(json.dumps({'m': d['module'], 'n': x['name'], 'k': x['kind'], 'cls': cls, 'b': bucket,
                                 'kw': kw, 'hdr': mod.raw[hs:he][:600], 'binders': x['binders'],
                                 'res': res, 'type': x['type'][:300], 'typed': typed},
                                ensure_ascii=False) + '\n')
    fo.close(); fs.close()
