#!/bin/bash
# usage: o3full-series.sh <pairs> <jobs list, e.g. "4 1"> <logdir>
set -uo pipefail

echo "target $(git -C "$T" rev-parse HEAD) $(cat "$T/lean-toolchain")"; git -C "$T" status --short

top -l 1 -o mem | head -15
echo "procs before:"; ps -axww -o pid,ppid,args | grep -E 'extract|litedoc4' | grep -v grep
cd "$T"
for j in $JOBS; do
  for i in $(seq 1 "$N"); do
    if [ $((i % 2)) = 1 ]; then order="O0 O3"; else order="O3 O0"; fi
    for lvl in $order; do
      tag="j$j-$lvl-$i"; d="$S/out-$tag"; rm -rf "$d"
      echo "=== run $tag"
      /usr/bin/time -l "$L4" build --root "$T" --lib InformationTheory --out "$d" --extractor-bin "$S/extract-$lvl" --jobs "$j" --timings "$LOG/$tag.timings.json" >"$LOG/$tag.log" 2>"$LOG/$tag.time"
      echo "rc=$?"
      grep -E '^(serve   ready|extract |build   full)' "$LOG/$tag.log"
      grep -E '^(total|analyze|linkIndex|writeIR) ' "$d/work/serve.out"
      grep -E 'real|maximum resident|page faults' "$LOG/$tag.time"
      cp "$d/work/serve.out" "$LOG/$tag.serve.out"; cp "$d/work/extract-timings-1.json" "$LOG/$tag.extract-timings.json"
    done
    a="$S/out-j$j-O0-$i"; b="$S/out-j$j-O3-$i"
    if /usr/bin/diff -r "$a/ir" "$b/ir" >/dev/null; then irs=identical; else irs=DIFFER; fi
    if cmp -s "$a/link-index.lidx" "$b/link-index.lidx"; then lx=identical; else lx=DIFFER; fi
    if /usr/bin/diff -r "$a/site" "$b/site" >/dev/null; then st=identical; else st=differs; fi
    echo "pair j$j-$i: ir/ ($(find "$a/ir" -type f | wc -l | tr -d ' ') files) $irs; link-index.lidx ($(stat -f %z "$a/link-index.lidx") B) $lx; site/ ($(find "$a/site" -type f | wc -l | tr -d ' ') files) $st"
    if [ "$i" = 1 ]; then
      if [ -n "${KEEP_J:-}" ]; then :; fi
      rm -rf "$S/keep-j$j-O0"; mv "$a" "$S/keep-j$j-O0"
    else rm -rf "$a"; fi
    rm -rf "$b"
  done
done
k=($S/keep-j*-O0)
if [ ${#k[@]} -ge 2 ]; then
  if /usr/bin/diff -r "${k[0]}/ir" "${k[1]}/ir" >/dev/null && cmp -s "${k[0]}/link-index.lidx" "${k[1]}/link-index.lidx"; then echo "across jobs (${k[0]##*/} vs ${k[1]##*/}): ir/ + lidx identical"; else echo "across jobs: DIFFER"; fi
fi
rm -rf $S/keep-j*-O0
echo "## after"
df -h /System/Volumes/Data | tail -1
top -l 1 -o mem | head -15
echo "procs after:"; ps -axww -o pid,ppid,args | grep -E 'extract|litedoc4' | grep -v grep
date
echo "series done"

    g = lambda pat, s: float(re.search(pat, s, re.M)[1])
    r = dict(j=j, lvl=lvl, i=i,
             ext=g(r'^extract \d+ module\(s\) in ([\d.]+) s', L),
             ready=g(r'^serve   ready (\d+)', L) / 1e9,
             req=g(r'^total\s+([\d.]+)s', V), ana=g(r'^analyze\s+([\d.]+)s', V), li=g(r'^linkIndex\s+([\d.]+)s', V),
             build=g(r'^build   full in ([\d.]+) s', L),
             wall=g(r'([\d.]+) real', T), cpu=g(r'([\d.]+) user', T) + g(r'([\d.]+) sys', T),
             rss=g(r'(\d+)\s+maximum resident', T) / 1e9, pf=int(g(r'(\d+)\s+page faults', T)))
    rows.append(r)
rows.sort(key=lambda r: (-r['j'], r['i'], r['lvl']))
cols = ['ext', 'ready', 'req', 'ana', 'li', 'build', 'wall', 'cpu', 'rss', 'pf']
print('jobs lvl run | extract-phase envload request analyze linkIndex | build-full wall user+sys peakRSS(GB) faults')
for r in rows:
    print(f"{r['j']:>4} {r['lvl']} {r['i']:>3} | {r['ext']:7.3f} {r['ready']:7.3f} {r['req']:7.3f} {r['ana']:7.3f} {r['li']:7.3f} | {r['build']:7.3f} {r['wall']:6.2f} {r['cpu']:6.2f} {r['rss']:6.3f} {r['pf']:>7}")
skip1 = len(sys.argv) > 2
print('\nmedian [min-max] per binary/jobs' + (' (run 1 excluded)' if skip1 else ''))
for j in sorted({r['j'] for r in rows}, reverse=True):
    med = {}
    for lvl in ('O0', 'O3'):
        sel = [r for r in rows if r['j'] == j and r['lvl'] == lvl and not (skip1 and r['i'] == 1)]
        med[lvl] = {c: st.median([r[c] for r in sel]) for c in cols}
        print(f"jobs {j} {lvl} n={len(sel)}: " + '  '.join(f"{c} {st.median([r[c] for r in sel]):.3f} [{min(r[c] for r in sel):.3f}-{max(r[c] for r in sel):.3f}]" for c in cols))
    for c in ('ext', 'req', 'ana', 'li', 'build', 'wall', 'cpu'):
        a, b = med['O0'][c], med['O3'][c]
        print(f"  jobs {j} gain {c}: {a:.3f} -> {b:.3f} = -{a-b:.3f} s, {a/b:.2f}x")
