===================================================================================================================================
Date: 2026-10-05. Context: docs/multiversion/plan.md D10 (one own .olean reader in the newest Lean; printing every
version with the newest Lean and Mathlib), U8 (every name-keyed lookup asked of an environment holding the newest

THE ONE SUBSTITUTION, AND EVERY REDIRECT TO OLD DATA
----------------------------------------------------
The hybrid arm runs `Litedoc4.run` of the same Extract.lean with `(env, world)` given, where the native arm imports. The
`World` fields and the extractor sites that now read them (native answer -> hybrid answer):
  moduleNames        header.moduleNames -> the old closure in importModulesCore's order (DFS post-order over the
                     8,170 roots in list order, from the decoded .olean imports): link-index `@` section and group
                     order, `loaded modules`.
  moduleIndex?       env.getModuleIdx? -> old index: candidate enumeration, module docs.
  constNames i       header.moduleData[i].constNames -> decoded old ModuleData (the private part, as v4.31.0's
                     finalizeImport takes it): the candidate list and the first-module `seen` rule, the link index walk.
  imports i          moduleData[i].imports -> decoded old imports.
  moduleOf?          env.getModuleIdxFor? -> old const2ModIdx rebuilt by v4.31.0's finalizeImport rule: modules in
                     order, each module's constNames, then the extraConstNames of its interpreter data (the `.ir` file
                     for a `module`, else the module's own), first module wins (1,344,766 names). Readers: refs ->
                     defining module (refs, deps/*.json), link-index ownership, tactic bucketing.
  contains           env.contains / env.find?.isSome -> old constant set (766,828): extOriginOf (`P.ext` for
                     `P.ext_iff`), declModifiers (`f._unsafe_rec` -> `partial`).
  moduleDocs         getModuleDoc? -> decoded old moduleDocExt entries per old module (8,981 modules of the closure
                     have any; 12,449 docs in the 8,035 target modules that do).
  docString?         findDocString? -> the same function in the hybrid when the old data holds the docstring (resolved
                     through the old inherit_doc map and the tactic-alternative map); a HARD STOP (counted, the text
                     `<hard stop: ...>` written instead) when the old data holds it only as Verso or not at all while
                     the running Lean's builtin table answers. Fired 0 times in this run.
  tactics            allTacticDocs -> HARD STOP: none, counted, printed as "NOT ANSWERED"; the tactic list comes from
                     the parser tables, which in the hybrid are the newest code's.
Sites not redirected, and why: `env.find? n` for a candidate or a link-index name n (n comes from the old lists and every
old constant is in the hybrid, replaced in place); `getEnv` for printing, telescopes, instances, attributes (they read
the merged extensions below); the diagnostics `--tactics-emulate`, `--tactics-probe`, `--dump-tactics` (not run; they
still read the environment's own tables).

(label, name), refs as (module, name) pairs, dependency entries by name, link-index lines by (module group, name).

def cmp(field, where, a, b):
    tot[field] += 1
    if a == b:
        eq[field] += 1
    else:
        note(field, where, a, b)


def reconcile(what, a_keys, b_keys, where):
    """Counts keys present on one side only; returns the keys present on both, in a's order."""
    sa, sb = set(a_keys), set(b_keys)
    for k in a_keys:
        if k not in sb:
            side['native-only ' + what] += 1
            note(what + '.presence', where, k, None)
    for k in b_keys:
        if k not in sa:
            side['hybrid-only ' + what] += 1
            note(what + '.presence', where, None, k)
    both = [k for k in a_keys if k in sb]
    tot[what + '.presence'] += len(sa | sb)
    eq[what + '.presence'] += len(both)
    return both


def compare_obj(prefix, where, a, b, skip=()):
    for k in sorted(set(a) | set(b)):

        cmp(prefix + k, where, a.get(k, '<absent>'), b.get(k, '<absent>'))

    for it in items:

        seen[k] += 1
        out.append((k if seen[k] == 1 else f'{k}#{seen[k]}', it))
    return out

            groups[cur].append((parts[0], tuple(parts[1:])))
        else:
            cur = line
            order.append(cur)
            groups[cur] = []
    return marker, at, groups, order

def print_class(a, b):

def tagged_consts(sp):
    out = Counter()

    def walk(x):
        if isinstance(x, list):
            if len(x) >= 4 and isinstance(x[0], int) and x[2] == 1 and isinstance(x[3], str):
                out[x[3]] += 1
            else:
                for y in x:
                    walk(y)
    walk(sp)
    return out


def span_only(a, b):
    lost = tagged_consts(a) - tagged_consts(b)
    gained = tagged_consts(b) - tagged_consts(a)
    if not gained and lost and set(lost) <= {'Filter.EventuallyEq', 'Filter.EventuallyLE'}:
        return 'S  =ᵐ[μ] / ≤ᵐ[μ] token carries no constant tag in the newest Mathlib (same text, one link fewer)'
    return 'S? span difference with the printed text equal'


def classify(field, r):
    a, b, w = r['native'], r['hybrid'], r['where']
    if field == 'decl.attrs':
        if s(a).replace('implicit_reducible', 'instance_reducible') == s(b):
            return 'R1 reducibility attribute: v4.31.0 spells it implicit_reducible, the running v4.34.1 instance_reducible'
        return 'R? other attribute difference'
    if field in ('decl.typeCode', 'decl.binderCode', 'decl.equationCode'):
        if text_diff_decls.get(w):
            return 'follows a printed-text difference of the same declaration'
        return span_only(a, b)
    if field in ('decl.refs', 'ref(pair).presence'):
        if text_diff_decls.get(w):
            return 'follows a printed-text difference of the same declaration'
        name = (a or b)[1] if field == 'ref(pair).presence' else None
        if field == 'ref(pair).presence' and name in ('Filter.EventuallyEq', 'Filter.EventuallyLE'):
            return 'S  =ᵐ[μ] / ≤ᵐ[μ] token carries no constant tag in the newest Mathlib (same text, one link fewer)'
        if field == 'decl.refs':
            gone = Counter(map(tuple, a)) - Counter(map(tuple, b))
            if not (Counter(map(tuple, b)) - Counter(map(tuple, a))) and {x[1] for x in gone} <= {'Filter.EventuallyEq', 'Filter.EventuallyLE'}:
                return 'S  =ᵐ[μ] / ≤ᵐ[μ] token carries no constant tag in the newest Mathlib (same text, one link fewer)'
        return 'E  references collected by an equation attempt that failed on both sides at different points'
    if field in ('member.code', 'member.binderCode'):
        return 'follows a printed-text difference of the same member' if w in member_text_diff else span_only(a, b)
    if field == 'decl.equations' and len(a) != len(b):
        return 'G  equation generation differs (newest generator on old definitions): number of equations differs'
    if field == 'equation(item)' and '<absent>' in (s(a), s(b)):
        return 'G  equation generation differs (newest generator on old definitions): number of equations differs'
    if field == 'equation(item)' and w.rsplit('#', 1)[0] in eq_count_diff:
        return 'G  equation generation differs (newest generator on old definitions): number of equations differs'
    if field == 'decl.members':
        ta, tb = [m.get('text') for m in a], [m.get('text') for m in b]
        ba, bb = [m.get('binders') for m in a], [m.get('binders') for m in b]
        if ta == tb and ba == bb:
            return span_only([m.get('code') for m in a] + [m.get('binderCode') for m in a],
                             [m.get('code') for m in b] + [m.get('binderCode') for m in b])
        return print_class(json.dumps([ta, ba], ensure_ascii=False), json.dumps([tb, bb], ensure_ascii=False))
    if field in ('decl.type', 'decl.binders', 'decl.equations', 'equation(item)', 'member.text',
                 'member.binders'):
        return print_class(a, b)
    if field in ('decl.doc', 'member.doc'):
        if '<hard stop' in s(b):
            return 'H  hard stop (value not produced from old data)'
        if s(a).replace('/doc/reference/4.31.0/', '/doc/reference/4.34.1/') == s(b):
            return 'M  manual link rewritten with the running Lean version (4.31.0 -> 4.34.1)'
        return 'docstring text differs'
    if field == 'declaration.presence':
        return 'A  declaration failed in the hybrid: an autoParam syntax constant has no IR in the newest environment'
    if field == 'decl.index':
        return 'I  position shifted by declarations missing earlier in the same module'
    if field in ('module.tactics', 'tactic.presence'):
        return 'T  hard stop: the tactic table is not answered from old data'
    return 'unclassified'


for field in sorted(by_field):

    for r in rs:

    print(f'== {field}: {len(rs)} not equal')

=== checks.py (derived index fields and the declaration index)
"""Consistency checks behind three derived index fields and the declaration index.
usage: checks.py <native-run-dir> <hybrid-run-dir> <details.jsonl>"""
import json, os, sys, filecmp
from collections import Counter, defaultdict
na, hy, det = sys.argv[1:4]
ia = {m['module']: m for m in json.load(open(os.path.join(na, 'ir/index.json')))['modules']}
ib = {m['module']: m for m in json.load(open(os.path.join(hy, 'ir/index.json')))['modules']}
same_bytes = same_hash = 0
bad = []
for m in ia:
    fa, fb = os.path.join(na, 'ir/modules', m + '.json'), os.path.join(hy, 'ir/modules', m + '.json')
    identical = filecmp.cmp(fa, fb, shallow=False)
    heq = ia[m]['contentHash'] == ib[m]['contentHash']
    same_bytes += identical
    same_hash += heq
    if identical != heq:
        bad.append(m)
print(f'module files byte-identical {same_bytes} of {len(ia)}; contentHash equal {same_hash}; hash equality disagrees with file equality: {len(bad)}')
missing = defaultdict(list)
idx_diff = []
for l in open(det):
    r = json.loads(l)
    if r['field'] == 'declaration.presence' and r['hybrid'] is None:
        missing[r['where']].append(r['native'])
    if r['field'] == 'decl.index':
        idx_diff.append(r)
explained = 0
cache = {}
for r in idx_diff:
    m, n = r['where'].split(':', 1)
    if m not in cache:
        A = json.load(open(os.path.join(na, 'ir/modules', m + '.json')))
        cache = {m: {d['name']: d['index'] for d in A['declarations']}}
    pos = cache[m]
    shift = sum(1 for x in missing.get(m, []) if pos[x] < r['native'])
    if r['native'] - r['hybrid'] == shift and shift > 0:
        explained += 1
print(f'decl.index differences: {len(idx_diff)}; equal to the number of earlier declarations missing on the hybrid side: {explained}')
print(f'modules with a missing declaration: {len(missing)}')

=== runx.sh (runs one program under /usr/bin/time -l with a workspace's `lake env` variables)
#!/bin/bash
# usage: runx.sh <lake-env-file> <cwd> <out-prefix> <program> [args...]
# Runs <program> under /usr/bin/time -l with the variables `lake env` set in <cwd>, so the
# timing wraps the program and not lake. stdout -> <out-prefix>.out, stderr -> <out-prefix>.err
set -uo pipefail
envf=$1; cwd=$2; out=$3; shift 3
for v in LEAN_PATH LEAN_SYSROOT LEAN_SRC_PATH LEAN_GITHASH PATH ELAN_TOOLCHAIN LEAN LAKE LAKE_HOME; do
  val=$(grep "^$v=" "$envf" | head -1 | cut -d= -f2-)
  export "$v=$val"
done
cd "$cwd" || exit 9
/usr/bin/time -l "$@" > "$out.out" 2> "$out.err"
rc=$?
echo "exit $rc" >> "$out.err"
exit $rc
