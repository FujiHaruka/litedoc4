THE WORKFLOW (to recreate the run on a branch of that name)
-----------------------------------------------------------
name: reader gates on Linux

on:
  push:
    branches: [reader-linux-tmp]

permissions:
  contents: read

jobs:
  reader:
    runs-on: ubuntu-latest
    timeout-minutes: 240
    env:
      LOGS: /home/runner/reader-linux/logs
      LAKE: /home/runner/.elan/bin/lake
    steps:
      - uses: actions/checkout@v7

      - name: install elan
        run: |
          curl -sSfL https://raw.githubusercontent.com/leanprover/elan/master/elan-init.sh \
            | sh -s -- -y --default-toolchain none
          echo "$HOME/.elan/bin" >> "$GITHUB_PATH"

      - name: record the machine
        run: |
          mkdir -p "$LOGS"
          benchmarks/tools/record-runner.sh "$LOGS/runner-start.txt"
          cat "$LOGS/runner-start.txt"
          df -h / /mnt 2>/dev/null || df -h /
          free -m

      - name: every row of lean-toolchains.txt and every writer the reader has a record for
        run: |
          set -euo pipefail
          {
            awk '{ sub(/#.*/, "") } NF { print $1 }' tools/lean-toolchains.txt
            sed -n 's/.*leanVersion := "\([0-9.]*\)".*/leanprover\/lean4:v\1/p' extractor/reader/OleanReader/Writers.lean
          } | sort -u > "$LOGS/toolchains.txt"
          cat "$LOGS/toolchains.txt"
          while read -r tc; do elan toolchain install "$tc"; done < "$LOGS/toolchains.txt"
          ls ~/.elan/toolchains
          df -h / /mnt 2>/dev/null || df -h /

      - name: litedoc4
        id: build
        run: tools/build-lean-exe.sh --toolchain-from e2e/micro

      - name: reader-oracle-gate
        if: ${{ !cancelled() && steps.build.outcome == 'success' }}
        env:
          READER_ORACLE_WORK: /home/runner/reader-linux/reader-oracle
        run: tools/reader-oracle-gate.sh --keep > "$LOGS/reader-oracle-gate.txt" 2>&1

      - name: reader-hybrid-gate
        if: ${{ !cancelled() && steps.build.outcome == 'success' }}
        env:
          READER_HYBRID_WORK: /home/runner/reader-linux/reader-hybrid
        run: tools/reader-hybrid-gate.sh --keep > "$LOGS/reader-hybrid-gate.txt" 2>&1

      - name: mv-reader-gate
        if: ${{ !cancelled() && steps.build.outcome == 'success' }}
        run: tools/mv-reader-gate.sh --out /home/runner/reader-linux/mv-reader --keep > "$LOGS/mv-reader-gate.txt" 2>&1

      - name: the gates' output
        if: always()
        run: |
          for f in reader-oracle-gate reader-hybrid-gate mv-reader-gate; do
            echo "##### $f"
            if [ -f "$LOGS/$f.txt" ]; then cat "$LOGS/$f.txt"; else echo "(no output)"; fi
          done
          df -h / /mnt 2>/dev/null || df -h /
          free -m

      - name: upload the logs
        if: always()
        uses: actions/upload-artifact@v7
        with:
          name: reader-linux
          retention-days: 14
          path: |
            /home/runner/reader-linux/logs
            /home/runner/reader-linux/mv-reader/logs
            /home/runner/reader-linux/mv-reader/times.txt
            /home/runner/reader-linux/reader-hybrid/**/*.txt
            /home/runner/reader-linux/reader-hybrid/**/*.log
            /home/runner/reader-linux/reader-oracle/**/*.log
            /home/runner/reader-linux/reader-oracle/**/*.err
            /home/runner/reader-linux/reader-oracle/*.txt
