title: Page not found
template: splash
editUrl: false
pagefind: false
hero:
  tagline: Nothing in this documentation is at this address. Try the search, or start from the home page.
  actions:
    - text: Home
      link: /litedoc4/
      icon: right-arrow
head:
  - tag: script
    content: |
      (() => {
        const sample = "/litedoc4/sample/";
        if (!location.pathname.startsWith(sample)) return;
        const root = document.documentElement;
        root.style.visibility = "hidden";
        const parsed = new Promise((done) => {
          if (document.readyState === "loading") addEventListener("DOMContentLoaded", done, { once: true });
          else done();
        });
        const page = fetch(`${sample}404.html`).then((r) => (r.ok ? r.text() : Promise.reject(r.status)));
        Promise.all([page, parsed]).then(
          ([html]) => {
            document.open();
            document.write(html);
            document.close();
          },
          () => {
            root.style.visibility = "";
          },
        );
      })();
